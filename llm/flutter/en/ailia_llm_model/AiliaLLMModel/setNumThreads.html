Sets the CPU thread count before open. nThreads is 0 for automatic selection, which is also the default. On Android, automatic selection uses performance (big/prime) cores, falling back to the existing core-count heuristic if topology is unavailable. Target decode/prefill, assistant and VLM/ALM CPU inference threads share this value. Throws an ArgumentError for values outside 0..INT_MAX, or an Exception if the model is already open or the native operation fails.

Implementation

void setNumThreads(int nThreads) {
  if (nThreads < 0 || nThreads > 0x7fffffff) {
    throw ArgumentError.value(nThreads, 'nThreads', 'must be between 0 and INT_MAX');
  }
  if (pLLm != nullptr && pLLm.value != nullptr) {
    final status = dllHandle.ailiaLLMSetNumThreads(pLLm.value, nThreads);
    if (status != ailia_llm_dart.AILIA_LLM_STATUS_SUCCESS) {
      throw Exception('ailiaLLMSetNumThreads returned an error status $status');
    }
  }
  _numThreads = nThreads;
}
