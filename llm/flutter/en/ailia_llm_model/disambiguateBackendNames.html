List<String> disambiguateBackendNames(List<String> names) {
  final counts = <String, int>{};
  for (final name in names) {
    counts[name] = (counts[name] ?? 0) + 1;
  }
  return [
    for (int i = 0; i < names.length; ++i)
      counts[names[i]]! > 1 ? '${names[i]} [$i]' : names[i],
  ];
}
